const sharp = require('./node_modules/sharp');
sharp('c:/real estate/logo.png').raw().toBuffer({ resolveWithObject: true }).then(async ({ data, info }) => {
  const w = info.width, h = info.height;
  const out = Buffer.alloc(w * h * 4);

  let bgCount = 0, fgCount = 0, aureliaCount = 0;

  for (let y = 0; y < h; y++) {
    for (let x = 0; x < w; x++) {
      const idx = (y * w + x) * 3;
      const oIdx = (y * w + x) * 4;

      // 1. Outside logo bounding box is 100% transparent:
      if (y < 25 || x < 180 || x > 1550 || y > 895) {
        out[oIdx] = 0; out[oIdx+1] = 0; out[oIdx+2] = 0; out[oIdx+3] = 0;
        bgCount++;
        continue;
      }

      const r = data[idx];
      const g = data[idx+1];
      const b = data[idx+2];

      const sat = Math.max(r, g, b) - Math.min(r, g, b);
      const lum = 0.299 * r + 0.587 * g + 0.114 * b;
      const warm = r - b;

      // Checkerboard background test:
      // White squares (lum >= 220)
      // Grey squares (lum >= 150, sat <= 20, warm <= 12)
      // Any neutral grey between elements (lum >= 115, sat <= 10, warm <= 10)
      const isCheckerboard = 
        (lum >= 220 && sat <= 12) ||
        (lum >= 150 && sat <= 22 && warm <= 12) ||
        (lum >= 115 && sat <= 10 && warm <= 8);

      if (isCheckerboard) {
        out[oIdx] = 0; out[oIdx+1] = 0; out[oIdx+2] = 0; out[oIdx+3] = 0;
        bgCount++;
      } else {
        fgCount++;
        let alpha = 1;

        // Antialiasing transition on edges
        if (lum >= 140 && warm <= 22) {
          const wF = Math.max(0, (warm - 8) / 14);
          const lF = Math.max(0, (150 - lum) / 25);
          alpha = Math.min(1, Math.max(wF, lF));
        }

        let cr = r, cg = g, cb = b;

        // Check if this pixel is part of the dark 'AURELIA' text on the left:
        // x in [260, 860], y in [170, 260], dark (lum < 150)
        const isAureliaDarkText = (x >= 260 && x <= 860 && y >= 170 && y <= 265 && lum < 155 && sat < 25);

        if (isAureliaDarkText) {
          aureliaCount++;
          // Invert/illuminate dark text to clean, crisp platinum white so it stands out with 100% clarity on dark intro 1:
          // The darker the original stroke, the brighter white it is!
          const strokeStrength = Math.max(0, Math.min(1, (155 - lum) / 120));
          // Light platinum / white with warm gold tint matching brand:
          cr = Math.round(250 * strokeStrength + 210 * (1 - strokeStrength));
          cg = Math.round(248 * strokeStrength + 205 * (1 - strokeStrength));
          cb = Math.round(242 * strokeStrength + 195 * (1 - strokeStrength));
          alpha = Math.max(alpha, strokeStrength);
        } else if (alpha < 0.95 && alpha > 0.05) {
          // De-halo normal edges
          cr = Math.min(255, Math.max(0, Math.round((r - (1 - alpha) * 215) / alpha)));
          cg = Math.min(255, Math.max(0, Math.round((g - (1 - alpha) * 215) / alpha)));
          cb = Math.min(255, Math.max(0, Math.round((b - (1 - alpha) * 220) / alpha)));
        }

        out[oIdx] = cr;
        out[oIdx+1] = cg;
        out[oIdx+2] = cb;
        out[oIdx+3] = Math.round(alpha * 255);
      }
    }
  }

  console.log('Background removed:', bgCount);
  console.log('Logo foreground:', fgCount);
  console.log('AURELIA text pixels illuminated to crisp platinum white:', aureliaCount);

  await sharp(out, { raw: { width: w, height: h, channels: 4 } })
    .trim()
    .png({ compressionLevel: 8 })
    .toFile('c:/real estate/luxury-real-estate/public/logo-clean.png');

  console.log('Saved master logo-clean.png');
});
